cat << 'EOF' > default.conf
server {
    listen 80;

    location / {
        root   /usr/share/nginx/html;
        index  index.html index.htm;
        try_files $uri $uri/ /index.html;
    }

    # This is the "bridge" that connects your frontend to your backend
    location /register {
        proxy_pass http://backend:3001/register;
    }

    location /login {
        proxy_pass http://backend:3001/login;
    }
}
EOF
